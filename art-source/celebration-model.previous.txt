import * as T from "three";
import { RoundedBoxGeometry } from "three/addons/geometries/RoundedBoxGeometry.js";
import type { StoryModel } from "./story-models";
import { createHumanCharacter } from "./human-character";

// Native 3D stage and locally adapted CC0 rigged characters. The phone records this same scene.
export function createCelebrationModel(
  viewportWidth: () => number = () =>
    typeof window !== "undefined" ? innerWidth : 1440,
): StoryModel {
  const group = new T.Group();
  const set = new T.Group();
  const creator = new T.Group();
  const phone = new T.Group();
  phone.name = "capture-phone";
  group.add(set, creator, phone);
  const materials: T.Material[] = [];
  const material = (color: string, metalness = 0, roughness = 0.8) => {
    const m = new T.MeshStandardMaterial({ color, metalness, roughness });
    materials.push(m);
    return m;
  };
  const ivory = material("#e9dfc9"),
    wall = material("#e2d7be"),
    cream = material("#fff2d6");
  const textile = (color: string, silk = false) => {
    const m = new T.MeshPhysicalMaterial({
      color,
      roughness: silk ? 0.78 : 0.95,
      sheen: silk ? 0.65 : 0.28,
      sheenColor: new T.Color(color).lerp(new T.Color("#e6dbc8"), 0.3),
      sheenRoughness: silk ? 0.6 : 0.9,
      side: T.DoubleSide,
    });
    materials.push(m);
    return m;
  };
  const gold = material("#aa8748", 0.72, 0.42),
    teal = textile("#426d64", true),
    forest = textile("#203e35");
  const amber = textile("#a67942", true),
    trousers = textile("#ad9d80"),
    jacket = textile("#b0b09d"),
    silk = textile("#d8c9ab", true),
    linen = textile("#ddd3bd");
  const dark = material("#1a211e", 0.15, 0.5),
    hair = material("#3c352e", 0, 0.96),
    shoe = material("#3b2c23");
  const eyeWhite = material("#c0b29e"),
    pupil = material("#4b4037");
  const skins = [
    material("#bd8969"),
    material("#cd9b79"),
    material("#ae7759"),
    material("#c18e6c"),
  ];
  skins.forEach((m) => {
    m.roughness = 0.68;
  });
  const leaf = material("#667953"),
    flower = material("#f8eacc");
  const glow = new T.MeshStandardMaterial({
    color: "#ffedb7",
    emissive: "#ffd385",
    emissiveIntensity: 1.2,
  });
  materials.push(glow);
  const mesh = (
    parent: T.Object3D,
    geometry: T.BufferGeometry,
    mat: T.Material,
    x = 0,
    y = 0,
    z = 0,
  ) => {
    const m = new T.Mesh(geometry, mat);
    m.position.set(x, y, z);
    m.castShadow = true;
    m.receiveShadow = true;
    parent.add(m);
    return m;
  };
  const box = (
    parent: T.Object3D,
    w: number,
    h: number,
    d: number,
    mat: T.Material,
    x = 0,
    y = 0,
    z = 0,
    r = 0.05,
  ) =>
    mesh(
      parent,
      new RoundedBoxGeometry(w, h, d, 2, Math.min(r, w / 3, h / 3, d / 3)),
      mat,
      x,
      y,
      z,
    );
  const ball = (
    parent: T.Object3D,
    mat: T.Material,
    x: number,
    y: number,
    z: number,
    sx: number,
    sy: number,
    sz: number,
  ) => {
    const m = mesh(parent, new T.SphereGeometry(1, 32, 24), mat, x, y, z);
    m.scale.set(sx, sy, sz);
    return m;
  };
  const cylinder = (
    parent: T.Object3D,
    rt: number,
    rb: number,
    h: number,
    mat: T.Material,
    x = 0,
    y = 0,
    z = 0,
  ) => mesh(parent, new T.CylinderGeometry(rt, rb, h, 32), mat, x, y, z);
  const tube = (
    parent: T.Object3D,
    points: number[][],
    radius: number,
    mat: T.Material,
  ) =>
    mesh(
      parent,
      new T.TubeGeometry(
        new T.CatmullRomCurve3(
          points.map((p) => new T.Vector3(...(p as [number, number, number]))),
        ),
        24,
        radius,
        6,
        false,
      ),
      mat,
    );
  const limb = (
    parent: T.Object3D,
    a: T.Vector3,
    b: T.Vector3,
    radius: number,
    mat: T.Material,
  ) => {
    const m = mesh(
      parent,
      new T.CylinderGeometry(radius, radius, a.distanceTo(b), 24),
      mat,
    );
    m.position.copy(a).add(b).multiplyScalar(0.5);
    m.quaternion.setFromUnitVectors(
      new T.Vector3(0, 1, 0),
      b.clone().sub(a).normalize(),
    );
    return m;
  };
  const v = (x: number, y: number, z: number) => new T.Vector3(x, y, z);

  // A single architectural space, open towards the viewer.
  const floor = box(set, 60, 0.12, 60, ivory, 0, -2.4, 0);
  floor.receiveShadow = true;
  box(set, 70, 40, 0.2, wall, 0, 17.65, -3.5);
  const backdrop = new T.Shape();
  backdrop.moveTo(-5.6, -2.34);
  backdrop.lineTo(-5.6, -0.1);
  backdrop.bezierCurveTo(-5.6, 5, 4.2, 5, 4.2, -0.1);
  backdrop.lineTo(4.2, -2.34);
  backdrop.closePath();
  mesh(
    set,
    new T.ExtrudeGeometry(backdrop, {
      depth: 0.12,
      bevelEnabled: true,
      bevelSegments: 3,
      bevelSize: 0.09,
      bevelThickness: 0.05,
      steps: 1,
    }),
    ivory,
    0,
    0,
    -3.22,
  );
  tube(
    set,
    [
      [-5.55, -2.25, -3.01],
      [-5.5, 0, -3.01],
      [-3.8, 2.8, -3.01],
      [-0.7, 3.7, -3.01],
      [2.4, 2.8, -3.01],
      [4.15, 0, -3.01],
      [4.15, -2.25, -3.01],
    ],
    0.025,
    cream,
  );
  // Recess-like ornamental niches kept behind the action.
  for (const [x, y] of [
    [-3.8, -0.3],
    [-2.7, 1.3],
  ]) {
    const niche = new T.Group();
    niche.position.set(x, y, -3.03);
    set.add(niche);
    box(niche, 0.65, 0.8, 0.06, wall, 0, 0, 0);
    ball(niche, wall, 0, 0.4, 0, 0.325, 0.325, 0.035);
    cylinder(niche, 0.14, 0.19, 0.32, cream, 0, -0.15, 0.17);
    const torus = mesh(
      niche,
      new T.TorusGeometry(0.14, 0.05, 8, 24),
      cream,
      0,
      0.03,
      0.17,
    );
    torus.rotation.x = 0.1;
  }
  cylinder(set, 0.014, 0.014, 1.7, gold, -0.8, 3.2, -0.9);
  const shade = mesh(
    set,
    new T.SphereGeometry(0.65, 32, 16, 0, Math.PI * 2, 0, Math.PI * 0.52),
    gold,
    -0.8,
    2.34,
    -0.9,
  );
  shade.scale.set(1, 0.55, 1);
  ball(set, glow, -0.8, 2.22, -0.9, 0.22, 0.1, 0.22);
  const pendant = new T.PointLight("#ffd99a", 4, 7, 2);
  pendant.position.set(-0.8, 2.05, -0.9);
  set.add(pendant);

  const characters: ReturnType<typeof createHumanCharacter>[] = [];
  const palette = (
    skin: T.Material,
    cloth: T.Material,
    pants: T.Material,
    sleeves?: T.Material,
  ) => ({
    skin,
    cloth,
    trousers: pants,
    sleeves,
    hair,
    gold,
    silk,
    shoes: shoe,
    eyes: eyeWhite,
    iris: pupil,
  });
  const seated = (
    x: number,
    z: number,
    female: boolean,
    cloth: T.Material,
    skin: T.Material,
    turn: number,
  ) => {
    const p = new T.Group();
    p.position.set(x, -0.65, z);
    p.rotation.y = turn;
    set.add(p);
    const furniture = new T.Group();
    furniture.position.y = 0.65;
    p.add(furniture);
    box(furniture, 0.7, 0.1, 0.67, cream, 0, -1.43, -0.04, 0.04);
    box(furniture, 0.65, 0.72, 0.09, ivory, 0, -0.97, -0.39, 0.04);
    for (const x of [-0.29, 0.29])
      for (const z of [-0.27, 0.24])
        limb(furniture, v(x, -1.47, z), v(x * 1.2, -2.3, z * 1.2), 0.022, gold);
    const character = createHumanCharacter(
      p,
      female,
      "seated",
      palette(skin, cloth, trousers, female ? undefined : linen),
    );
    characters.push(character);
    return { p, face: character.head };
  };
  const womanA = seated(-2.65, 0.25, true, teal, skins[0], 0.6);
  const womanB = seated(-1.45, -0.55, true, amber, skins[1], 0.24);
  const guestC = seated(0.4, -0.05, false, jacket, skins[3], -0.55);
  const toast = new T.Group();
  toast.position.set(-0.65, 0.15, -1.25);
  set.add(toast);
  const toastCharacter = createHumanCharacter(
    toast,
    false,
    "toast",
    palette(skins[2], forest, linen),
  );
  characters.push(toastCharacter);
  const toastHead = toastCharacter.head;
  const glassHolder = new T.Group();
  toast.add(glassHolder);
  const glassMaterial = new T.MeshPhysicalMaterial({
    color: "#fff4d8",
    transparent: true,
    opacity: 0.3,
    roughness: 0.15,
    metalness: 0.1,
    side: T.DoubleSide,
  });
  materials.push(glassMaterial);
  const drink = new T.MeshPhysicalMaterial({
    color: "#ac7439",
    transparent: true,
    opacity: 0.58,
    roughness: 0.2,
  });
  materials.push(drink);
  const wineglass = (parent: T.Object3D, x: number, y: number, z: number) => {
    cylinder(parent, 0.085, 0.1, 0.014, glassMaterial, x, y, z);
    cylinder(parent, 0.009, 0.009, 0.17, glassMaterial, x, y + 0.09, z);
    cylinder(parent, 0.005, 0.005, 0.17, gold, x, y + 0.09, z);
    mesh(
      parent,
      new T.LatheGeometry(
        [
          new T.Vector2(0, 0),
          new T.Vector2(0.07, 0.025),
          new T.Vector2(0.095, 0.15),
          new T.Vector2(0.08, 0.25),
        ],
        24,
      ),
      glassMaterial,
      x,
      y + 0.18,
      z,
    );
    cylinder(parent, 0.081, 0.05, 0.1, drink, x, y + 0.26, z);
  };
  wineglass(glassHolder, 0, 0, 0);

  // Round table and cake are the common focal point in both viewpoints.
  const tableX = -1.1,
    tableZ = 0.63;
  cylinder(set, 1.55, 1.55, 0.12, cream, tableX, -0.57, tableZ).name =
    "celebration-tabletop";
  cylinder(set, 0.24, 0.55, 1.69, ivory, tableX, -1.46, tableZ);
  cylinder(set, 0.43, 0.43, 0.04, gold, tableX, -0.47, tableZ);
  cylinder(set, 0.36, 0.36, 0.27, cream, tableX, -0.32, tableZ);
  const icing = new T.MeshStandardMaterial({
    color: "#fff5df",
    roughness: 0.9,
  });
  materials.push(icing);
  for (const y of [-0.44, -0.18]) {
    const rim = mesh(
      set,
      new T.TorusGeometry(0.354, 0.008, 8, 64),
      icing,
      tableX,
      y,
      tableZ,
    );
    rim.rotation.x = Math.PI / 2;
  }
  for (let i = 0; i < 64; i++) {
    const a = (i / 64) * Math.PI * 2;
    cylinder(
      set,
      0.003,
      0.003,
      0.23,
      icing,
      tableX + Math.cos(a) * 0.358,
      -0.32,
      tableZ + Math.sin(a) * 0.358,
    );
  }
  for (const x of [-0.17, 0, 0.17]) {
    cylinder(set, 0.009, 0.009, 0.18, gold, tableX + x, -0.07, tableZ);
    ball(set, glow, tableX + x, 0.035, tableZ, 0.013, 0.026, 0.013);
  }
  for (const [dx, dz] of [
    [-0.85, -0.05],
    [0.9, -0.05],
    [-0.25, -0.68],
  ]) {
    cylinder(set, 0.19, 0.19, 0.022, ivory, tableX + dx, -0.492, tableZ + dz);
    wineglass(set, tableX + dx + 0.14, -0.477, tableZ + dz + 0.12);
  }
  cylinder(set, 0.12, 0.16, 0.35, ivory, tableX - 0.4, -0.31, tableZ - 0.42);
  for (let i = 0; i < 7; i++) {
    const a = i * 2.4,
      fx = tableX - 0.4 + Math.cos(a) * 0.19,
      fy = 0.015 + Math.sin(i) * 0.075,
      fz = tableZ - 0.42 + Math.sin(a) * 0.14;
    tube(
      set,
      [
        [tableX - 0.4, -0.24, tableZ - 0.42],
        [fx, fy - 0.05, fz],
      ],
      0.009,
      leaf,
    );
    for (let j = 0; j < 5; j++) {
      const b = (j / 5) * Math.PI * 2;
      const petal = ball(
        set,
        flower,
        fx + Math.cos(b) * 0.043,
        fy + Math.sin(b) * 0.043,
        fz,
        0.03,
        0.054,
        0.009,
      );
      petal.rotation.z = b - Math.PI / 2;
      petal.rotation.y = Math.sin(i + j) * 0.35;
    }
    ball(set, gold, fx, fy, fz + 0.01, 0.009, 0.009, 0.004);
    const l = ball(set, leaf, fx - 0.06, fy - 0.13, fz, 0.065, 0.025, 0.018);
    l.rotation.z = 0.7;
  }

  // A rigged person holds the phone in the original foreground position.
  creator.position.set(2.9, 0.4, 3.2);
  creator.scale.setScalar(1.3);
  creator.rotation.y = 0.48;
  const creatorCharacter = createHumanCharacter(
    creator,
    false,
    "creator",
    palette(skins[1], jacket, dark),
  );
  characters.push(creatorCharacter);

  const target = new T.WebGLRenderTarget(768, 1344, {
    depthBuffer: true,
    type: T.HalfFloatType,
  });
  target.texture.colorSpace = T.LinearSRGBColorSpace;
  const screenMaterial = new T.MeshBasicMaterial({
    map: target.texture,
    toneMapped: true,
  });
  materials.push(screenMaterial);
  box(phone, 0.8, 1.58, 0.095, dark, 0, 0, 0, 0.055);
  mesh(
    phone,
    new T.PlaneGeometry(0.722, 1.32),
    screenMaterial,
    0,
    0.015,
    0.052,
  );
  box(phone, 0.72, 0.09, 0.006, dark, 0, 0.713, 0.056, 0.002);
  ball(phone, pupil, 0, 0.725, 0.065, 0.026, 0.026, 0.009);
  box(phone, 0.24, 0.018, 0.01, cream, 0, -0.73, 0.059, 0.004);
  const shutter = mesh(
    phone,
    new T.TorusGeometry(0.046, 0.008, 8, 24),
    cream,
    0,
    -0.68,
    0.065,
  );
  const recordDot = ball(
    phone,
    material("#c7583e"),
    -0.29,
    0.69,
    0.065,
    0.018,
    0.018,
    0.008,
  );
  const timeline = box(phone, 0.56, 0.012, 0.01, gold, 0, -0.62, 0.064, 0.003);
  const recordingCamera = new T.PerspectiveCamera(35, 512 / 896, 0.1, 100);
  recordingCamera.name = "recording-camera";
  group.add(recordingCamera);
  recordingCamera.position.set(-1.1, 0.8, 12.5);
  recordingCamera.lookAt(-1.1, -0.1, 0);
  let current = 0;
  let replayAt: number | undefined;
  const smooth = (p: number, a: number, b: number) =>
    T.MathUtils.smoothstep(p, a, b);
  const mix = T.MathUtils.lerp;
  group.userData.materials = materials;
  return {
    group,
    update(p, time) {
      current = p;
      const pull = smooth(p, 0.42, 0.55),
        focus = smooth(p, 0.55, 0.8),
        finish = smooth(p, 0.88, 1),
        mobile = viewportWidth() < 651;
      creator.position.set(
        mobile ? 2.1 : 2.9,
        mobile ? 0.08 : 0.4,
        mobile ? 3.1 : 3.2,
      );
      creator.scale.setScalar(mobile ? 1.14 : 1.3);
      let replay = p > 0.8 ? smooth(p, 0.81, 0.94) : smooth(p, 0.08, 0.33);
      if (replayAt !== undefined) {
        replay = smooth(time - replayAt, 0, 1.5);
        if (time - replayAt > 1.6) replayAt = undefined;
      }
      toastCharacter.animateToast(replay);
      glassHolder.position
        .copy(toastCharacter.localPosition("finger2-1.R"))
        .add(v(0.035, -0.1, 0.015));
      toastHead.rotation.y = mix(-0.2, 0.05, replay);
      womanA.face.rotation.y = mix(0.02, 0.26, replay);
      womanB.face.rotation.y = mix(0.04, 0.24, replay);
      guestC.face.rotation.y = mix(-0.12, -0.3, replay);
      phone.position.set(
        mix(mobile ? 1.8 : 2.56, mobile ? 0 : -1.16, focus),
        mix(mobile ? 0.764 : 1.18, mobile ? 0.82 : 0.02, focus),
        mix(mobile ? 2.523 : 2.54, 4.8, pull),
      );
      phone.rotation.y = mix(0.48, 0, pull);
      phone.rotation.z = mix(-0.02, 0, focus);
      phone.scale.setScalar(
        mix(mobile ? 0.798 : 0.91, mobile ? 1.16 : 2.15, focus),
      );
      set.visible = p < 0.82;
      creator.visible = p < 0.79;
      shutter.visible = p < 0.84;
      recordDot.visible = p < 0.84;
      timeline.scale.x = Math.max(0.02, replay);
      timeline.position.x = (timeline.scale.x - 1) * 0.28;
      group.userData.recordingProgress = replay;
      group.userData.phase =
        p < 0.35 ? "occasion" : p < 0.8 ? "recording" : "replay";
      group.userData.copyVisible = finish > 0.2;
    },
    updateCamera(camera, distance, aspect, p) {
      const focus = smooth(p, 0.55, 0.8),
        orbit = smooth(p, 0.1, 0.42),
        mobile = aspect < 0.8;
      const startX = mobile ? -2.8 : -3.5;
      const finalDistance = mobile
        ? 6.7
        : Math.max(
            6.7,
            4.3 / (2 * Math.tan(T.MathUtils.degToRad(16)) * aspect),
          );
      camera.position.set(
        mix(startX + orbit * 0.8, 0, focus),
        mix(2.1, 0.15, focus),
        mix(distance, finalDistance + 4.8, focus),
      );
      camera.lookAt(
        mix(-0.15, mobile ? 0 : 0, focus),
        mix(-0.25, 0, focus),
        mix(0.1, 4.8, focus),
      );
    },
    beforeRender(renderer, scene) {
      const setVisible = set.visible,
        creatorVisible = creator.visible,
        phoneVisible = phone.visible;
      set.visible = true;
      creator.visible = false;
      phone.visible = false;
      const previous = renderer.getRenderTarget();
      renderer.setRenderTarget(target);
      renderer.render(scene, recordingCamera);
      renderer.setRenderTarget(previous);
      set.visible = setVisible;
      creator.visible = creatorVisible;
      phone.visible = phoneVisible;
      group.userData.progress = current;
    },
    dispose() {
      target.dispose();
      characters.forEach((character) => character.dispose());
    },
    replay(time) {
      replayAt = time;
    },
  };
}
